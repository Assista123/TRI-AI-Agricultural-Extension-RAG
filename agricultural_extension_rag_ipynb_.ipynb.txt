        "\n",
        "class BM25Okapi:\n",
        "    def __init__(self, corpus, k1=1.5, b=0.75):\n",
        "        self.k1 = k1\n",
        "        self.b = b\n",
        "        self.corpus_size = len(corpus)\n",
        "        self.avgdl = sum(len(doc) for doc in corpus) / self.corpus_size if self.corpus_size else 0\n",
        "        self.doc_freqs = []\n",
        "        self.idf = {}\n",
        "        self.doc_len = []\n",
        "\n",
        "        df = Counter()\n",
        "        for doc in corpus:\n",
        "            self.doc_len.append(len(doc))\n",
        "            frequencies = Counter(doc)\n",
        "            self.doc_freqs.append(frequencies)\n",
        "            for word in frequencies:\n",
        "                df[word] += 1\n",
        "\n",
        "        for word, freq in df.items():\n",
        "            self.idf[word] = math.log((self.corpus_size - freq + 0.5) / (freq + 0.5) + 1)\n",
        "\n",
        "    def get_scores(self, query):\n",
        "        scores = [0.0] * self.corpus_size\n",
        "        for word in query:\n",
        "            if word not in self.idf:\n",
        "                continue\n",
        "            idf = self.idf[word]\n",
        "            for i, doc_freqs in enumerate(self.doc_freqs):\n",
        "                if word in doc_freqs:\n",
        "                    freq = doc_freqs[word]\n",
        "                    numerator = idf * freq * (self.k1 + 1)\n",
        "                    denominator = freq + self.k1 * (1 - self.b + self.b * (self.doc_len[i] / self.avgdl))\n",
        "                    scores[i] += numerator / denominator\n",
        "        return scores"

        "\n",
        "AGRI_DICTIONARY = {\n",
        "    \"yellow\": \"chlorosis leaf yellowing nutrient deficiency nitrogen\",\n",
        "    \"spots\": \"lesions leaf spot fungal blight leaf disease\",\n",
        "    \"curling\": \"leaf curl viral infection aphids drought stress\",\n",
        "    \"rot\": \"root rot stalk rot dampening off decay bacterial softness\",\n",
        "    \"bugs\": \"pests insects fall armyworm stem borer whiteflies vector\",\n",
        "    \"maize\": \"corn zea mays cereal grain crop\",\n",
        "    \"cassava\": \"manihot esculenta tuber cassava mosaic virus\",\n",
        "    \"rice\": \"paddy oryza sativa blast disease panicle\",\n",
        "    \"fertilizer\": \"NPK urea nitrogen phosphorus potassium soil amendment\",\n",
        "    \"water\": \"irrigation drought moisture stress rainfall deficit\",\n",
        "}\n",
        "\n",
        "def expand_agri_query(query: str) -> str:\n",
        "    tokens = re.findall(r\"\\b\\w+\\b\", query.lower())\n",
        "    expanded = set(tokens)\n",
        "    for token in tokens:\n",
        "        if token in AGRI_DICTIONARY:\n",
        "            expanded.update(AGRI_DICTIONARY[token].split())\n",
        "    return \" \".join(expanded)\n",
        "\n",
        "def generate_hyde_doc(query: str, expanded_query: str) -> str:\n",
        "    return (\n",
        "        f\"Technical Agricultural Extension Guide regarding {query}.\\n\"\n",
        "        f\"Diagnosis and Management Key Terms: {expanded_query}.\\n\"\n",
        "        f\"Recommended agronomic practices include crop monitoring, proper soil nutrient management, \"\n",
        "        f\"pest control protocols, and application of recommended treatments for affected crops.\"\n",
        "    )"

        "# Lexical index\n",
        "tokenized_corpus = [doc.lower().split() for doc in doc_texts]\n",
        "bm25_index = BM25Okapi(tokenized_corpus)\n",

        "\n",
        "BGE_PREFIX = \"Represent this sentence for searching relevant passages: \"\n",
        "submission_rows = []\n",
        "\n",
        "for q_id, raw_q in zip(test_queries_df[query_id_col], test_queries_df[query_text_col].fillna(\"\").astype(str)):\n",
        "    expanded_q = expand_agri_query(raw_q)\n",
        "    hyde_passage = generate_hyde_doc(raw_q, expanded_q)\n",
        "\n",
        "    # Lexical\n",
        "    bm25_scores = bm25_index.get_scores(expanded_q.lower().split())\n",
        "    top_bm25 = np.argsort(bm25_scores)[::-1][:60]\n",
        "\n",
        "    # Dense (HyDE)\n",
        "    q_emb = retriever.encode(BGE_PREFIX + hyde_passage, convert_to_tensor=True, normalize_embeddings=True)\n",
        "    dense_scores = torch.matmul(doc_embeddings, q_emb).cpu().numpy()\n",
        "    top_dense = np.argsort(dense_scores)[::-1][:60]\n",
        "\n",
        "    # RRF\n",
        "    rrf = {}\n",
        "    for rank, idx in enumerate(top_bm25):\n",
        "        rrf[idx] = rrf.get(idx, 0.0) + 1.0 / (60 + rank + 1)\n",
        "    for rank, idx in enumerate(top_dense):\n",
        "        rrf[idx] = rrf.get(idx, 0.0) + 1.0 / (60 + rank + 1)\n",
        "    candidates = sorted(rrf, key=rrf.get, reverse=True)[:30]\n",
        "\n",
        "    # Dual rerank\n",
        "    pairs = [[raw_q, doc_texts[i]] for i in candidates]\n",
        "    with torch.autocast(device_type=device, dtype=torch.float16 if device == \"cuda\" else torch.float32):\n",
        "        s_bge = reranker_bge.predict(pairs, batch_size=32, show_progress_bar=False)\n",
        "        s_mxbai = reranker_mxbai.predict(pairs, batch_size=32, show_progress_bar=False)\n",
        "\n",
        "    norm_bge = (s_bge - s_bge.min()) / (np.ptp(s_bge) + 1e-8)\n",
        "    norm_mxbai = (s_mxbai - s_mxbai.min()) / (np.ptp(s_mxbai) + 1e-8)\n",
        "    ensemble = 0.5 * norm_bge + 0.5 * norm_mxbai\n",
        "\n",
        "    top10 = [candidates[i] for i in np.argsort(ensemble)[::-1][:10]]\n",
        "    for doc_idx in top10:\n",
        "        submission_rows.append({\n",
        "            \"QueryId\": q_id,\n",
        "            \"DocumentId\": documents_df.iloc[doc_idx][doc_id_col]\n",
        "        })\n",
        "\n",
        "print(f\"Done. Generated {len(submission_rows)} rows.\")"
